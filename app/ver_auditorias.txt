📌 Consulta 1: Ver cambios de precios
sql
SELECT
    DATE_FORMAT(a.fecha, '%d/%m/%Y %H:%i') AS cuando,
    u.nombre AS quien,
    p.nombre AS producto,
    JSON_EXTRACT(a.datos_anteriores, '$.precio') AS precio_viejo,
    JSON_EXTRACT(a.datos_nuevos, '$.precio') AS precio_nuevo
FROM AuditoriaAcciones a
JOIN Producto p ON a.id_registro = p.id_producto
LEFT JOIN Usuario u ON a.id_usuario = u.id_usuario
WHERE a.accion = 'MODIFICAR_PRODUCTO'
  AND a.tabla_afectada = 'Producto'
  AND JSON_EXTRACT(a.datos_nuevos, '$.precio') IS NOT NULL
ORDER BY a.fecha DESC
LIMIT 50;
